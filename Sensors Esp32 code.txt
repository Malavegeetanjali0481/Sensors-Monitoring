#include <WiFi.h>
#include <HTTPClient.h>
#include <ThingSpeak.h>
#include <DHT.h>

/* WIFI */

char ssid[] = "ESP32TEST";
char pass[] = "12345678";

/* THINGSPEAK */

unsigned long channelNumber = 3393380;
const char * writeAPIKey = "P616VZAY3GRI1VIS";

WiFiClient client;

/* DHT22 */

#define DHTPIN 14
#define DHTTYPE DHT22

DHT dht(DHTPIN, DHTTYPE);

/* SENSOR PINS */

#define IR_PIN 27
#define FLAME_PIN 26
#define RAIN_PIN 34
#define SOUND_PIN 25
#define SOIL_PIN 35

/* LED */

#define LED_PIN 2

bool sensorsEnabled = false;

void setup()
{
  Serial.begin(115200);

  pinMode(IR_PIN, INPUT);
  pinMode(FLAME_PIN, INPUT);
  pinMode(SOUND_PIN, INPUT);

  pinMode(LED_PIN, OUTPUT);

  digitalWrite(LED_PIN, LOW);

  dht.begin();

  Serial.println("Connecting WiFi...");

  WiFi.begin(ssid, pass);

  while(WiFi.status() != WL_CONNECTED)
  {
    delay(1000);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi Connected");

  Serial.print("IP: ");
  Serial.println(WiFi.localIP());

  ThingSpeak.begin(client);
}

void loop()
{
  /* WIFI CHECK */

  if(WiFi.status() != WL_CONNECTED)
  {
    Serial.println("WiFi Lost");

    WiFi.disconnect();

    WiFi.begin(ssid, pass);

    while(WiFi.status() != WL_CONNECTED)
    {
      delay(1000);
      Serial.print(".");
    }

    Serial.println();
    Serial.println("WiFi Reconnected");
  }

  /* READ FIELD8 */

  HTTPClient http;

  String url =
  "https://api.thingspeak.com/channels/3393380/fields/8/last.txt";

  http.begin(url);

  int httpCode = http.GET();

  if(httpCode > 0)
  {
    String response = http.getString();

    response.trim();

    Serial.print("Field8 Response: ");
    Serial.println(response);

    if(response == "1")
    {
      sensorsEnabled = true;
    }
    else
    {
      sensorsEnabled = false;
    }
  }
  else
  {
    sensorsEnabled = false;

    Serial.print("HTTP Error: ");
    Serial.println(httpCode);
  }

  http.end();

  /* OFF MODE */

  if(!sensorsEnabled)
  {
    digitalWrite(LED_PIN, LOW);

    Serial.println("Sensors OFF");
    Serial.println("----------------");

    delay(5000);

    return;
  }

  /* ON MODE */

  digitalWrite(LED_PIN, HIGH);

  float temperature =
  dht.readTemperature();

  float humidity =
  dht.readHumidity();

  if(isnan(temperature) || isnan(humidity))
  {
    Serial.println("DHT Error");

    delay(2000);

    return;
  }

  int irValue =
  digitalRead(IR_PIN);

  int flameValue =
  digitalRead(FLAME_PIN);

  int soundValue =
  digitalRead(SOUND_PIN);

  int rainValue =
  analogRead(RAIN_PIN);

  int soilValue =
  analogRead(SOIL_PIN);

  Serial.println("-------------------------");

  Serial.print("Temperature: ");
  Serial.print(temperature);
  Serial.println(" °C");

  Serial.print("Humidity: ");
  Serial.print(humidity);
  Serial.println(" %");

  Serial.print("IR Sensor: ");
  Serial.println(irValue);

  Serial.print("Flame Sensor: ");
  Serial.println(flameValue);

  Serial.print("Rain Sensor: ");
  Serial.println(rainValue);

  Serial.print("Sound Sensor: ");
  Serial.println(soundValue);

  Serial.print("Soil Moisture: ");
  Serial.println(soilValue);

  /* UPLOAD */

  ThingSpeak.setField(1, temperature);
  ThingSpeak.setField(2, humidity);
  ThingSpeak.setField(3, irValue);
  ThingSpeak.setField(4, flameValue);
  ThingSpeak.setField(5, rainValue);
  ThingSpeak.setField(6, soundValue);
  ThingSpeak.setField(7, soilValue);

  int result =
  ThingSpeak.writeFields(
  channelNumber,
  writeAPIKey
  );

  if(result == 200)
  {
    Serial.println("Data Uploaded Successfully");
  }
  else
  {
    Serial.print("Upload Failed: ");
    Serial.println(result);
  }

  delay(20000);
}
